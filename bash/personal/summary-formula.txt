
Arithmetics:
sum=$((x+y))
or 
if [ $((i % 2)) -ne 0 ]

Conditionals:
1. if [ $x -ge $y ]; then
	echo....
   elif [...]; then
	echo...
   else; then
	echo...
   fi

2. for ((i=1; i<100; i++)); do
	echo...
   done

3. while ...; do
   done

4. switch or case
	case $1 in 
		Pattern1)
		echo"..."
		;;
		
		Pattern2)
		echo"..."
		;;

		Pattern3 | Pattern4)   --> either 3 or 4, same echo
		echo "..."
		;;
		
		*)           --> else/default
		echo
		;;
	esac      	--> inverse of case

5. printf "%.3f\n" $(echo "$sum / $n" | bc -l)
	--> This is how to escape Bash truncation and produce decimals
	--> Use bc -l as you calculate and early mention printf

6. Calling read for the first time calls first line
--> subsequent calls resume where you left from
--> in "read n" read got to stdin and gives stdin to n
